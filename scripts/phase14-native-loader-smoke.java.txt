public final class NativeLoaderSmoke {
  private NativeLoaderSmoke() {}

  public static void main(String[] args) {
    if (args.length != 1 || !args[0].startsWith("/")) {
      throw new IllegalArgumentException("expected one absolute native-library path");
    }
    dev.mantle.internal.NativeLoader.load(args[0]);
  }
}
